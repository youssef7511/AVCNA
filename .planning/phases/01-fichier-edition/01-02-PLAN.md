```csharp
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using FuzzySharp;
using Microsoft.EntityFrameworkCore;
using AVCNDB.WPF.Contracts.Services;
using AVCNDB.WPF.DAL;
using AVCNDB.WPF.Models;

namespace AVCNDB.WPF.Services;

/// <summary>
/// ML-based unknown data detection using FuzzySharp token-sort-ratio.
/// Threshold >= 80 = known, < 80 = unknown.
/// </summary>
public class FuzzyDetectionService : IUnknownDataDetectionService
{
    private readonly AppDbContext _context;

    /// <summary>Minimum score to consider a value "known" in the library</summary>
    public const int KnownThreshold = 80;

    public FuzzyDetectionService(AppDbContext context)
    {
        _context = context;
    }

    // ─── IUnknownDataDetectionService ────────────────────────────────────

    public async Task<DetectionReport> DetectAsync(EditionRow row)
    {
        var library = await LoadLibraryAsync();
        return BuildReport(row, library);
    }

    public async Task<List<DetectionReport>> DetectBatchAsync(IEnumerable<EditionRow> rows)
    {
        // Load library once for the whole batch
        var library = await LoadLibraryAsync();
        return rows.Select(r => BuildReport(r, library)).ToList();
    }

    public FieldDetectionResult CheckValue(string fieldName, string value, IEnumerable<string> libraryValues)
    {
        // Empty or whitespace value is always considered "known" (nothing to detect)
        if (string.IsNullOrWhiteSpace(value))
        {
            return new FieldDetectionResult
            {
                FieldName = fieldName,
                ImportedValue = value,
                IsKnown = true,
                Score = 100,
                BestMatch = null
            };
        }

        var libraryList = libraryValues.Where(v => !string.IsNullOrWhiteSpace(v)).ToList();

        if (libraryList.Count == 0)
        {
            // No library entries to compare against — treat as unknown
            return new FieldDetectionResult
            {
                FieldName = fieldName,
                ImportedValue = value,
                IsKnown = false,
                Score = 0,
                BestMatch = null
            };
        }

        // Find the best match using FuzzySharp token-sort-ratio
        string? bestMatch = null;
        int bestScore = 0;

        foreach (var libraryValue in libraryList)
        {
            int score = Fuzz.TokenSortRatio(value.ToUpperInvariant(), libraryValue.ToUpperInvariant());
            if (score > bestScore)
            {
                bestScore = score;
                bestMatch = libraryValue;
            }
        }

        return new FieldDetectionResult
        {
            FieldName = fieldName,
            ImportedValue = value,
            BestMatch = bestMatch,
            Score = bestScore,
            IsKnown = bestScore >= KnownThreshold
        };
    }

    // ─── Internals ────────────────────────────────────────────────────────

    private async Task<LibrarySnapshot> LoadLibraryAsync()
    {
        var dcis        = await _context.Dcis.Select(d => d.dciname ?? string.Empty).ToListAsync();
        var labos       = await _context.Laboss.Select(l => l.laboname ?? string.Empty).ToListAsync();
        var families    = await _context.Familiess.Select(f => f.familyname ?? string.Empty).ToListAsync();
        var formes      = await _context.Formess.Select(f => f.formname ?? string.Empty).ToListAsync();
        var voies       = await _context.Voiess.Select(v => v.voiename ?? string.Empty).ToListAsync();
        var specialites = await _context.Specialitess.Select(s => s.specialitename ?? string.Empty).ToListAsync();

        return new LibrarySnapshot(dcis, labos, families, formes, voies, specialites);
    }

    private DetectionReport BuildReport(EditionRow row, LibrarySnapshot lib)
    {
        var report = new DetectionReport();

        report.FieldResults.Add(CheckValue("Dci",             row.Dci,             lib.Dcis));
        report.FieldResults.Add(CheckValue("DciAssociation",  row.DciAssociation,  lib.Dcis));
        report.FieldResults.Add(CheckValue("Labo",            row.Labo,            lib.Labos));
        report.FieldResults.Add(CheckValue("Fam1",            row.Fam1,            lib.Families));
        report.FieldResults.Add(CheckValue("Fam2",            row.Fam2,            lib.Families));
        report.FieldResults.Add(CheckValue("Fam3",            row.Fam3,            lib.Families));
        report.FieldResults.Add(CheckValue("Forme",           row.Forme,           lib.Formes));
        report.FieldResults.Add(CheckValue("Voie",            row.Voie,            lib.Voies));
        report.FieldResults.Add(CheckValue("Specialite",      row.Specialite,      lib.Specialites));

        return report;
    }

    /// <summary>Immutable snapshot of library data for a batch run</summary>
    private record LibrarySnapshot(
        List<string> Dcis,
        List<string> Labos,
        List<string> Families,
        List<string> Formes,
        List<string> Voies,
        List<string> Specialites);
}
```

```csharp
using System.Collections.Generic;
using System.Data;
using System.Linq;
using System.Threading.Tasks;
using Microsoft.EntityFrameworkCore;
using AVCNDB.WPF.Contracts.Services;
using AVCNDB.WPF.DAL;
using AVCNDB.WPF.Models;

namespace AVCNDB.WPF.Services;

/// <summary>
/// Implementation of IEditionFileService.
/// Orchestrates Excel import, library validation, row approval, and export.
/// </summary>
public class EditionFileService : IEditionFileService
{
    private readonly AppDbContext _context;
    private readonly IUnknownDataDetectionService _detectionService;
    private readonly IExcelService _excelService;

    public EditionFileService(
        AppDbContext context,
        IUnknownDataDetectionService detectionService,
        IExcelService excelService)
    {
        _context = context;
        _detectionService = detectionService;
        _excelService = excelService;
    }

    // ─── Import ──────────────────────────────────────────────────────────

    public async Task<ImportResult> ImportExcelAsync(string filePath, EditionSourceType sourceType = EditionSourceType.ExcelSimple)
    {
        try
        {
            var dataTable = await _excelService.ReadExcelAsync(filePath);
            var rows = MapDataTableToEditionRows(dataTable);

            return new ImportResult
            {
                Success = true,
                Rows = rows
            };
        }
        catch (Exception ex)
        {
            return new ImportResult
            {
                Success = false,
                ErrorMessage = ex.Message,
                Rows = new List<EditionRow>()
            };
        }
    }

    // ─── Validation ──────────────────────────────────────────────────────

    public async Task<List<EditionRow>> ValidateAgainstLibraryAsync(List<EditionRow> rows)
    {
        var reports = await _detectionService.DetectBatchAsync(rows);

        for (int i = 0; i < rows.Count; i++)
        {
            var row = rows[i];
            var report = reports[i];

            row.UnknownFields = report.UnknownFieldNames;

            if (report.HasUnknownFields)
                row.ActionFlag = ActionFlag.AjouterNew;
            else if (row.ActionFlag == ActionFlag.AjouterNew)
                row.ActionFlag = ActionFlag.Affecte; // was unknown, now resolved
        }

        return rows;
    }

    // ─── Approve / Reject ────────────────────────────────────────────────

    public async Task<bool> ApproveRowAsync(EditionRow row, bool insertMedic = true)
    {
        try
        {
            // Add unknown field values to library tables
            foreach (var fieldName in row.UnknownFields)
            {
                await AddToLibraryAsync(fieldName, row);
            }

            // Optionally insert/update the Medic record
            if (insertMedic)
            {
                var medic = MapEditionRowToMedic(row);
                if (row.ExistingMedicId.HasValue)
                {
                    var existing = await _context.Medics.FindAsync(row.ExistingMedicId.Value);
                    if (existing != null)
                    {
                        UpdateMedicFromRow(existing, row);
                    }
                }
                else
                {
                    await _context.Medics.AddAsync(medic);
                }
            }

            await _context.SaveChangesAsync();

            // Mark row as resolved
            row.UnknownFields.Clear();
            row.ActionFlag = ActionFlag.Affecte;
            row.RowStatus = RowStatus.Modified;

            return true;
        }
        catch
        {
            return false;
        }
    }

    public async Task RejectRowAsync(EditionRow row)
    {
        row.ActionFlag = ActionFlag.Desaffecte;
        row.UnknownFields.Clear();
        row.RowStatus = RowStatus.Active;
        await Task.CompletedTask;
    }

    // ─── Export ──────────────────────────────────────────────────────────

    public async Task ExportEditionFileAsync(IEnumerable<EditionRow> rows, string outputPath)
    {
        var rowList = rows.ToList();
        await _excelService.ExportToExcelAsync(rowList, outputPath, "Fichier d'édition");
    }

    // ─── Session ─────────────────────────────────────────────────────────

    public async Task<EditionFileSession> SaveSessionAsync(
        string sourceFilePath,
        string sourceType,
        int totalRows,
        int unknownRows)
    {
        var session = new EditionFileSession
        {
            sourcefilepath = sourceFilePath,
            sourcetype = sourceType,
            totalrows = totalRows,
            unknownrows = unknownRows,
            status = "Draft"
        };

        await _context.EditionFileSessions.AddAsync(session);
        await _context.SaveChangesAsync();

        return session;
    }

    // ─── Private helpers ─────────────────────────────────────────────────

    private static List<EditionRow> MapDataTableToEditionRows(DataTable dt)
    {
        var rows = new List<EditionRow>();
        int lineNo = 1;

        // Column name mapping: normalize header names to EditionRow properties
        // Supports both French column names (from reference app) and direct field names
        var colMap = BuildColumnMap(dt.Columns.Cast<DataColumn>().Select(c => c.ColumnName).ToList());

        foreach (DataRow dr in dt.Rows)
        {
            var row = new EditionRow { LineNumber = lineNo++ };

            row.PctCode     = GetString(dr, colMap, "pctcode", "Code PCT", "Code Pct");
            row.Denomination = GetString(dr, colMap, "itemname", "Dénomination", "Denomination");
            row.Dci         = GetString(dr, colMap, "dci1", "D.C.J", "DCI", "Dci");
            row.DciAssociation = GetString(dr, colMap, "dci2", "Association", "DCI Association");
            row.Forme       = GetString(dr, colMap, "forme", "Forme");
            row.Tableau     = GetString(dr, colMap, "tableau", "Tab.", "Tableau");
            row.Veic        = GetString(dr, colMap, "veic", "VEiC");
            row.Labo        = GetString(dr, colMap, "labo", "Labo.", "Labo");
            row.Designation = GetString(dr, colMap, "shortname", "Désignation", "Designation");
            row.Specialite  = GetString(dr, colMap, "specialite", "Spécialité");
            row.Fam1        = GetString(dr, colMap, "fam1", "Famille", "Fam1");
            row.Voie        = GetString(dr, colMap, "voie", "Voie");

            // Numeric fields
            if (decimal.TryParse(GetString(dr, colMap, "price", "Prix"), out var prix))
                row.Prix = prix;
            if (decimal.TryParse(GetString(dr, colMap, "refprice", "Prix-Réf", "Prix Réf"), out var prixRef))
                row.PrixRef = prixRef;

            // Boolean fields
            var rembStr = GetString(dr, colMap, "isremboursable", "Remb.", "Remboursable");
            row.IsRemboursable = rembStr is "1" or "oui" or "true" or "O";

            var apStr = GetString(dr, colMap, "isap", "A.P.", "AP");
            row.IsAP = apStr is "1" or "oui" or "true" or "O";

            rows.Add(row);
        }

        return rows;
    }

    private static Dictionary<string, int> BuildColumnMap(List<string> columns)
    {
        var map = new Dictionary<string, int>(StringComparer.OrdinalIgnoreCase);
        for (int i = 0; i < columns.Count; i++)
            map[columns[i].Trim()] = i;
        return map;
    }

    private static string GetString(DataRow dr, Dictionary<string, int> colMap, params string[] candidateNames)
    {
        foreach (var name in candidateNames)
        {
            if (colMap.TryGetValue(name, out int idx))
                return dr[idx]?.ToString()?.Trim() ?? string.Empty;
        }
        return string.Empty;
    }

    private async Task AddToLibraryAsync(string fieldName, EditionRow row)
    {
        switch (fieldName)
        {
            case "Dci" or "DciAssociation":
                var dciValue = fieldName == "Dci" ? row.Dci : row.DciAssociation;
                if (!string.IsNullOrWhiteSpace(dciValue))
                {
                    var exists = await _context.Dcis.AnyAsync(d => d.dciname == dciValue);
                    if (!exists)
                        await _context.Dcis.AddAsync(new Dci { dciname = dciValue });
                }
                break;

            case "Labo":
                if (!string.IsNullOrWhiteSpace(row.Labo))
                {
                    var exists = await _context.Laboss.AnyAsync(l => l.laboname == row.Labo);
                    if (!exists)
                        await _context.Laboss.AddAsync(new Labos { laboname = row.Labo });
                }
                break;

            case "Fam1" or "Fam2" or "Fam3":
                var famValue = fieldName == "Fam1" ? row.Fam1 : fieldName == "Fam2" ? row.Fam2 : row.Fam3;
                if (!string.IsNullOrWhiteSpace(famValue))
                {
                    var exists = await _context.Familiess.AnyAsync(f => f.familyname == famValue);
                    if (!exists)
                        await _context.Familiess.AddAsync(new Families { familyname = famValue });
                }
                break;

            case "Forme":
                if (!string.IsNullOrWhiteSpace(row.Forme))
                {
                    var exists = await _context.Formess.AnyAsync(f => f.formname == row.Forme);
                    if (!exists)
                        await _context.Formess.AddAsync(new Formes { formname = row.Forme });
                }
                break;

            case "Voie":
                if (!string.IsNullOrWhiteSpace(row.Voie))
                {
                    var exists = await _context.Voiess.AnyAsync(v => v.voiename == row.Voie);
                    if (!exists)
                        await _context.Voiess.AddAsync(new Voies { voiename = row.Voie });
                }
                break;

            case "Specialite":
                if (!string.IsNullOrWhiteSpace(row.Specialite))
                {
                    var exists = await _context.Specialitess.AnyAsync(s => s.specialitename == row.Specialite);
                    if (!exists)
                        await _context.Specialitess.AddAsync(new Specialites { specialitename = row.Specialite });
                }
                break;
        }
    }

    private static Medic MapEditionRowToMedic(EditionRow row) => new Medic
    {
        pctcode = row.PctCode,
        itemname = row.Denomination,
        dci1 = row.Dci,
        labo = row.Labo,
        forme = row.Forme,
        tableau = row.Tableau,
        veic = row.Veic,
        price = row.Prix,
        refprice = row.PrixRef,
        isremboursable = row.IsRemboursable,
        isap = row.IsAP,
        shortname = row.Designation,
        specialite = row.Specialite,
        fam1 = row.Fam1,
        fam2 = row.Fam2,
        fam3 = row.Fam3,
        voie = row.Voie
    };

    private static void UpdateMedicFromRow(Medic medic, EditionRow row)
    {
        medic.pctcode = row.PctCode;
```
