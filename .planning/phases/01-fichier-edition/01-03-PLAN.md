```csharp
using CommunityToolkit.Mvvm.ComponentModel;

namespace AVCNDB.WPF.ViewModels;

/// <summary>
/// Observable wrapper around EditionRow.
/// Exposes per-field IsUnknown_* booleans for XAML DataTrigger cell coloring (D-03).
/// </summary>
public partial class EditionRowViewModel : ObservableObject
{
    public EditionRow Row { get; }

    public EditionRowViewModel(EditionRow row)
    {
        Row = row;
        row.PropertyChanged += (_, e) =>
        {
            // Bubble property changes from the underlying EditionRow
            OnPropertyChanged(e.PropertyName);
            OnPropertyChanged(nameof(IsUnknown_Dci));
            OnPropertyChanged(nameof(IsUnknown_DciAssociation));
            OnPropertyChanged(nameof(IsUnknown_Labo));
            OnPropertyChanged(nameof(IsUnknown_Forme));
            OnPropertyChanged(nameof(IsUnknown_Fam1));
            OnPropertyChanged(nameof(IsUnknown_Fam2));
            OnPropertyChanged(nameof(IsUnknown_Fam3));
            OnPropertyChanged(nameof(IsUnknown_Voie));
            OnPropertyChanged(nameof(IsUnknown_Specialite));
            OnPropertyChanged(nameof(ActionLabel));
            OnPropertyChanged(nameof(HasAction));
            OnPropertyChanged(nameof(HasUnknownFields));
        };
    }

    // ─── Pass-through properties (for DataGrid column binding) ────────────

    public int LineNumber       => Row.LineNumber;
    public ActionFlag ActionFlag => Row.ActionFlag;
    public string ActionLabel   => Row.ActionLabel;
    public bool HasAction       => Row.HasAction;
    public bool HasUnknownFields => Row.HasUnknownFields;
    public bool IsSelected
    {
        get => Row.IsSelected;
        set { Row.IsSelected = value; OnPropertyChanged(); }
    }

    public string PctCode       => Row.PctCode;
    public string Denomination  => Row.Denomination;
    public string Dci           => Row.Dci;
    public string DciAssociation => Row.DciAssociation;
    public string Forme         => Row.Forme;
    public string Tableau       => Row.Tableau;
    public string Veic          => Row.Veic;
    public string Labo          => Row.Labo;
    public decimal PrixRef      => Row.PrixRef;
    public decimal Prix         => Row.Prix;
    public bool IsRemboursable  => Row.IsRemboursable;
    public bool IsAP            => Row.IsAP;

    public string Fam1          => Row.Fam1;
```

```csharp
using System.Collections.ObjectModel;
using System.Linq;
using System.Windows;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using AVCNDB.WPF.Contracts.Services;
using AVCNDB.WPF.Models;

namespace AVCNDB.WPF.ViewModels;

/// <summary>
/// ViewModel for the "Fichier d'édition" page (D-01, D-02, D-03, D-04, D-05, D-06).
/// Orchestrates Excel import, ML detection, row approval/rejection, and display filtering.
/// </summary>
public partial class EditionFileViewModel : ViewModelBase
{
    private readonly IEditionFileService _editionFileService;
    private readonly IDialogService _dialogService;

    // ─── Source rows (all) ────────────────────────────────────────────────
    private List<EditionRowViewModel> _allRows = new();

    // ─── Displayed rows (filtered) ────────────────────────────────────────
    [ObservableProperty]
    private ObservableCollection<EditionRowViewModel> _rows = new();

    [ObservableProperty]
    private EditionRowViewModel? _selectedRow;

    // ─── Import state ────────────────────────────────────────────────────
    [ObservableProperty]
    private string _currentFilePath = string.Empty;

    [ObservableProperty]
    private int _totalRowCount;

    [ObservableProperty]
    private int _unknownRowCount;

    [ObservableProperty]
    private int _approvedRowCount;

    // ─── Display filter ──────────────────────────────────────────────────
    /// <summary>
    /// "Tous" | "Actifs" | "Inactifs" | "Affectés" | "Non-affectés" |
    /// "Méd.[V.E.I]" | "Avec A.P" | "Sans A.P" | "Remboursés" | "Non-remboursés" |
    /// "Sélectionnés" | "Non-sélectionnés" | "Changements de prix"
    /// </summary>
    [ObservableProperty]
    private string _filterType = "Tous";

    // ─── Constructor ─────────────────────────────────────────────────────

    public EditionFileViewModel(
        IEditionFileService editionFileService,
        IDialogService dialogService)
    {
        _editionFileService = editionFileService;
        _dialogService = dialogService;
    }

    // ─── Property change handlers ─────────────────────────────────────────

    partial void OnFilterTypeChanged(string value)
    {
        ApplyFilter();
    }

    // ─── Commands ─────────────────────────────────────────────────────────

    /// <summary>
    /// Open file dialog → import Excel → run ML detection → populate grid
    /// </summary>
    [RelayCommand]
    private async Task ImportAsync()
    {
        // Open file picker (falls back to Win32 OpenFileDialog if IDialogService lacks ShowOpenFileDialog)
        string? filePath = null;

        try
        {
            // Try IDialogService first (it may or may not have ShowOpenFileDialog)
            var dialogType = _dialogService.GetType();
            var method = dialogType.GetMethod("ShowOpenFileDialog");
            if (method != null)
            {
                filePath = method.Invoke(_dialogService, new object[] { "Excel Files|*.xlsx;*.xls", "Importer un fichier" }) as string;
            }
        }
        catch { /* fall through to OpenFileDialog */ }

        if (string.IsNullOrEmpty(filePath))
        {
            // Direct Win32 dialog (guaranteed to work)
            var dialog = new Microsoft.Win32.OpenFileDialog
            {
                Filter = "Excel Files|*.xlsx;*.xls",
                Title = "Importer un fichier d'édition"
            };
            if (dialog.ShowDialog() != true)
                return;
            filePath = dialog.FileName;
        }

        await ExecuteAsync(async () =>
        {
            // 1. Import Excel
            var importResult = await _editionFileService.ImportExcelAsync(filePath);

            if (!importResult.Success)
            {
                await _dialogService.ShowErrorAsync("Erreur d'import", importResult.ErrorMessage ?? "Erreur inconnue");
                return;
            }

            // 2. Run ML detection
            var validatedRows = await _editionFileService.ValidateAgainstLibraryAsync(importResult.Rows);

            // 3. Wrap in ViewModels
            _allRows = validatedRows.Select(r => new EditionRowViewModel(r)).ToList();

            // 4. Update stats
            CurrentFilePath = filePath;
            TotalRowCount = _allRows.Count;
            UnknownRowCount = _allRows.Count(r => r.HasUnknownFields);
            ApprovedRowCount = 0;

            // 5. Apply filter and refresh grid
            ApplyFilter();

            // 6. Save session to DB
            await _editionFileService.SaveSessionAsync(
                filePath, "ExcelSimple", TotalRowCount, UnknownRowCount);

            StatusMessage = $"Import terminé : {TotalRowCount} lignes, {UnknownRowCount} inconnues.";
        }, "Importation en cours...");
    }

    /// <summary>
    /// Approve the selected row: add unknown values to library, insert Medic if needed
    /// </summary>
    [RelayCommand(CanExecute = nameof(CanExecuteRowAction))]
    private async Task ApproveRowAsync()
    {
        if (SelectedRow == null) return;

        var confirmed = await _dialogService.ShowConfirmAsync(
            "Approuver la ligne",
            $"Ajouter les données inconnues de '{SelectedRow.Denomination}' à la bibliothèque et insérer le médicament ?");

        if (!confirmed) return;

        await ExecuteAsync(async () =>
        {
            var success = await _editionFileService.ApproveRowAsync(SelectedRow.Row, insertMedic: true);
            if (success)
            {
                ApprovedRowCount++;
                UnknownRowCount = _allRows.Count(r => r.HasUnknownFields);
                ApplyFilter();
                await _dialogService.ShowSuccessAsync("Approuvé", $"'{SelectedRow.Denomination}' ajouté avec succès.");
            }
            else
            {
                await _dialogService.ShowErrorAsync("Erreur", "Impossible d'approuver cette ligne.");
            }
        }, "Approbation...");
    }

    /// <summary>
    /// Reject the selected row: clear unknown fields, set Désaffecté
    /// </summary>
    [RelayCommand(CanExecute = nameof(CanExecuteRowAction))]
    private async Task RejectRowAsync()
    {
        if (SelectedRow == null) return;

        await ExecuteAsync(async () =>
        {
            await _editionFileService.RejectRowAsync(SelectedRow.Row);
            UnknownRowCount = _allRows.Count(r => r.HasUnknownFields);
            ApplyFilter();
        }, "Rejet...");
    }

    /// <summary>
    /// Mark selected row as new (user decision, not ML-based)
    /// </summary>
    [RelayCommand(CanExecute = nameof(CanExecuteRowAction))]
    private void MarkAsNew()
    {
        if (SelectedRow == null) return;
        SelectedRow.Row.ActionFlag = ActionFlag.Nouveau;
    }

    /// <summary>
    /// Mark selected row for deletion
    /// </summary>
    [RelayCommand(CanExecute = nameof(CanExecuteRowAction))]
    private void MarkForDeletion()
    {
        if (SelectedRow == null) return;
        SelectedRow.Row.ActionFlag = ActionFlag.MarquerSuppression;
    }

    /// <summary>
    /// Reset selected row to its original imported state
    /// </summary>
    [RelayCommand(CanExecute = nameof(CanExecuteRowAction))]
    private void ResetRow()
    {
        if (SelectedRow == null) return;
        SelectedRow.Row.ActionFlag = ActionFlag.Reinitialiser;
    }

    /// <summary>
    /// Open the library manager dialog (D-05: dialog, not separate page)
    /// </summary>
    [RelayCommand]
    private async Task OpenLibraryManagerAsync()
    {
        // The library manager dialog is shown via DialogService.
        // The actual dialog window is registered separately (Plan 04 will wire the XAML).
        // For now, show a placeholder info until Plan 04 creates the dialog.
        await _dialogService.ShowInfoAsync(
            "Gestion de la bibliothèque",
            "Le gestionnaire de bibliothèque sera disponible ici. (À implémenter en Plan 04)");
    }

    /// <summary>
    /// Similarity search for a selected row
    /// </summary>
    [RelayCommand(CanExecute = nameof(CanExecuteRowAction))]
    private async Task SimilaritySearchAsync()
    {
        if (SelectedRow == null) return;
        await _dialogService.ShowInfoAsync(
            "Recherche de médicament similaire",
            $"Recherche pour : {SelectedRow.Denomination}");
    }

    /// <summary>
    /// Export current filtered rows to Excel
    /// </summary>
    [RelayCommand]
    private async Task ExportAsync()
    {
        var dialog = new Microsoft.Win32.SaveFileDialog
        {
            Filter = "Excel Files|*.xlsx",
            FileName = $"Fichier_edition_{DateTime.Now:yyyyMMdd}",
            Title = "Exporter le fichier d'édition"
        };
        if (dialog.ShowDialog() != true) return;

        await ExecuteAsync(async () =>
        {
            var rowsToExport = Rows.Select(vm => vm.Row).ToList();
            await _editionFileService.ExportEditionFileAsync(rowsToExport, dialog.FileName);
            await _dialogService.ShowSuccessAsync("Export réussi", $"Fichier exporté vers {dialog.FileName}");
        }, "Export en cours...");
    }

    // ─── Can-execute helpers ──────────────────────────────────────────────

    private bool CanExecuteRowAction() => SelectedRow != null;

    // ─── Filter logic ─────────────────────────────────────────────────────

    private void ApplyFilter()
    {
        var filtered = FilterType switch
        {
            "Actifs"          => _allRows.Where(r => r.Row.RowStatus == RowStatus.Active),
            "Inactifs"        => _allRows.Where(r => r.Row.RowStatus == RowStatus.Inactive),
            "Affectés"        => _allRows.Where(r => r.Row.ActionFlag == ActionFlag.Affecte),
            "Non-affectés"    => _allRows.Where(r => r.Row.ActionFlag == ActionFlag.AjouterNew || r.Row.ActionFlag == ActionFlag.None),
            "Méd.[V.E.I]"     => _allRows.Where(r => !string.IsNullOrWhiteSpace(r.Veic)),
            "Avec A.P"        => _allRows.Where(r => r.IsAP),
            "Sans A.P"        => _allRows.Where(r => !r.IsAP),
            "Remboursés"      => _allRows.Where(r => r.IsRemboursable),
            "Non-remboursés"  => _allRows.Where(r => !r.IsRemboursable),
            "Sélectionnés"    => _allRows.Where(r => r.IsSelected),
            "Non-sélectionnés" => _allRows.Where(r => !r.IsSelected),
            "Changements de prix" => _allRows.Where(r => r.Prix != r.PrixRef),
            _                 => _allRows // "Tous"
        };

        Rows = new ObservableCollection<EditionRowViewModel>(filtered);
    }
}
```

```csharp
using System.Collections.Generic;
using AVCNDB.WPF.Contracts.Services;
using AVCNDB.WPF.Models;
using AVCNDB.WPF.ViewModels;
using Moq;

namespace AVCNDB.WPF.Tests.ViewModels;

public class EditionFileViewModelTests
{
    private EditionFileViewModel CreateVm(
        Mock<IEditionFileService>? editionMock = null,
        Mock<IDialogService>? dialogMock = null)
    {
        editionMock ??= new Mock<IEditionFileService>();
        dialogMock ??= new Mock<IDialogService>();
        return new EditionFileViewModel(editionMock.Object, dialogMock.Object);
    }

    [Fact]
    public void Constructor_InitializesDefaults()
    {
        var vm = CreateVm();

        vm.Should().NotBeNull();
        vm.Rows.Should().BeEmpty();
        vm.FilterType.Should().Be("Tous");
        vm.SelectedRow.Should().BeNull();
        vm.TotalRowCount.Should().Be(0);
        vm.UnknownRowCount.Should().Be(0);
    }

    [Fact]
    public void Commands_AreNotNull()
    {
        var vm = CreateVm();

        vm.ImportCommand.Should().NotBeNull();
        vm.ApproveRowCommand.Should().NotBeNull();
        vm.RejectRowCommand.Should().NotBeNull();
        vm.MarkAsNewCommand.Should().NotBeNull();
        vm.MarkForDeletionCommand.Should().NotBeNull();
        vm.ResetRowCommand.Should().NotBeNull();
        vm.OpenLibraryManagerCommand.Should().NotBeNull();
        vm.SimilaritySearchCommand.Should().NotBeNull();
        vm.ExportCommand.Should().NotBeNull();
    }

    [Fact]
    public void FilterType_ChangeTo_Affectes_UpdatesRows()
    {
        var vm = CreateVm();

        // Manually populate _allRows via reflection (testing filter logic in isolation)
        var row1 = new EditionRowViewModel(new EditionRow { ActionFlag = ActionFlag.Affecte });
        var row2 = new EditionRowViewModel(new EditionRow { ActionFlag = ActionFlag.AjouterNew });

        // Use reflection to set the private field
        var allRowsField = typeof(EditionFileViewModel)
            .GetField("_allRows", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance)!;
        allRowsField.SetValue(vm, new List<EditionRowViewModel> { row1, row2 });

        // Act
        vm.FilterType = "Affectés";

        // Assert: only Affecte row visible
        vm.Rows.Should().HaveCount(1);
        vm.Rows[0].ActionFlag.Should().Be(ActionFlag.Affecte);
    }

    [Fact]
    public void FilterType_Tous_ShowsAllRows()
    {
        var vm = CreateVm();

        var allRowsField = typeof(EditionFileViewModel)
            .GetField("_allRows", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance)!;
        allRowsField.SetValue(vm, new List<EditionRowViewModel>
        {
            new(new EditionRow { ActionFlag = ActionFlag.Affecte }),
            new(new EditionRow { ActionFlag = ActionFlag.AjouterNew }),
            new(new EditionRow { ActionFlag = ActionFlag.None })
        });

        vm.FilterType = "Tous";

        vm.Rows.Should().HaveCount(3);
    }
}
```
