```csharp
using System.Collections.Generic;
using System.ComponentModel;
using System.ComponentModel.DataAnnotations.Schema;
using System.Runtime.CompilerServices;

namespace AVCNDB.WPF.Models;

/// <summary>
/// Action flag for an edition row (what the user/system wants to do with this row)
/// </summary>
public enum ActionFlag
{
    None = 0,
    AjouterNew = 1,       // "A ajouter" — row contains unknown data that must be added to library
    MarquerSuppression = 2, // "A supprimer" — row marked for deletion from medic
    Nouveau = 3,           // "Nouveau" — user manually marked as new
    Reinitialiser = 4,     // "Ré-initialiser" — reset to original imported state
    Affecte = 5,           // "Affecté" — all unknown fields resolved
    Desaffecte = 6         // "Désaffecté" — previously affecté, now reverted
}

/// <summary>
/// Status of a row after processing
/// </summary>
public enum RowStatus
{
    Active = 0,
    Inactive = 1,
    New = 2,
    Modified = 3
}

/// <summary>
/// Represents a single row in the Fichier d'édition DataGrid.
/// In-memory only — not persisted directly to DB.
/// Mirrors Medic columns + adds edition-specific metadata.
/// </summary>
public class EditionRow : INotifyPropertyChanged
{
    // ─── Edition metadata ────────────────────────────────────────────────
    private bool _isSelected;
    public bool IsSelected
    {
        get => _isSelected;
        set { _isSelected = value; OnPropertyChanged(); }
    }

    /// <summary>Line number in the imported file (1-based)</summary>
    public int LineNumber { get; set; }

    private ActionFlag _actionFlag = ActionFlag.None;
    public ActionFlag ActionFlag
    {
        get => _actionFlag;
        set { _actionFlag = value; OnPropertyChanged(); OnPropertyChanged(nameof(ActionLabel)); OnPropertyChanged(nameof(HasAction)); }
    }

    private RowStatus _rowStatus = RowStatus.Active;
    public RowStatus RowStatus
    {
        get => _rowStatus;
        set { _rowStatus = value; OnPropertyChanged(); }
    }

    /// <summary>
    /// Fields that the ML detection found to be unknown (below threshold).
    /// Used by the UI to color those specific cells blue+bold.
    /// </summary>
    public List<string> UnknownFields { get; set; } = new();

    /// <summary>True if this row has any unknown fields</summary>
    public bool HasUnknownFields => UnknownFields.Count > 0;

    /// <summary>True if ActionFlag is not None</summary>
    public bool HasAction => ActionFlag != ActionFlag.None;

    /// <summary>Human-readable label for the Action column badge</summary>
    public string ActionLabel => ActionFlag switch
    {
        ActionFlag.AjouterNew => "A ajouter",
        ActionFlag.MarquerSuppression => "A supprimer",
        ActionFlag.Nouveau => "Nouveau",
        ActionFlag.Reinitialiser => "Ré-initialisé",
        ActionFlag.Affecte => "Affecté",
        ActionFlag.Desaffecte => "Désaffecté",
        _ => string.Empty
    };

    // ─── Medic mirror columns (match Medic.cs field names exactly) ────────

    /// <summary>Existing medic recordid if matched, null if new</summary>
    public int? ExistingMedicId { get; set; }

    public string PctCode { get; set; } = string.Empty;        // Code PCT
    public string Denomination { get; set; } = string.Empty;   // Dénomination (itemname)
    public string Dci { get; set; } = string.Empty;            // D.C.J (dci1, primary DCI)
    public string DciAssociation { get; set; } = string.Empty; // dci2+dci3+dci4 combined
    public string Forme { get; set; } = string.Empty;          // Forme
    public string Tableau { get; set; } = string.Empty;        // Tab.
    public string Veic { get; set; } = string.Empty;           // VEiC
    public string Labo { get; set; } = string.Empty;           // Labo.
    public decimal PrixRef { get; set; }                        // Prix-Réf
    public decimal Prix { get; set; }                           // Prix
    public bool IsRemboursable { get; set; }                   // Remb.
    public bool IsAP { get; set; }                             // A.P.
    public string Designation { get; set; } = string.Empty;   // shortname
    public string Specialite { get; set; } = string.Empty;
    public string Fam1 { get; set; } = string.Empty;
    public string Fam2 { get; set; } = string.Empty;
    public string Fam3 { get; set; } = string.Empty;
    public string Voie { get; set; } = string.Empty;

    // ─── INotifyPropertyChanged ──────────────────────────────────────────
    public event PropertyChangedEventHandler? PropertyChanged;
    protected void OnPropertyChanged([CallerMemberName] string? name = null)
        => PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(name));
}
```

```csharp
using System.Collections.Generic;
using System.Threading.Tasks;
using AVCNDB.WPF.Models;

namespace AVCNDB.WPF.Contracts.Services;

/// <summary>
/// Result for a single field's detection against the library
/// </summary>
public class FieldDetectionResult
{
    /// <summary>Field name (e.g. "Dci", "Labo", "Forme")</summary>
    public string FieldName { get; set; } = string.Empty;

    /// <summary>The imported value being evaluated</summary>
    public string ImportedValue { get; set; } = string.Empty;

    /// <summary>Best matching library value (null if no library entries)</summary>
    public string? BestMatch { get; set; }

    /// <summary>FuzzySharp token-sort-ratio score (0–100)</summary>
    public int Score { get; set; }

    /// <summary>True if score >= threshold (default 80) — value is known</summary>
    public bool IsKnown { get; set; }
}

/// <summary>
/// Full detection report for one EditionRow (all fields checked)
/// </summary>
public class DetectionReport
{
    public List<FieldDetectionResult> FieldResults { get; set; } = new();

    /// <summary>Names of all fields that are unknown (IsKnown = false)</summary>
    public List<string> UnknownFieldNames => FieldResults
        .Where(r => !r.IsKnown && !string.IsNullOrWhiteSpace(r.ImportedValue))
        .Select(r => r.FieldName)
        .ToList();

    /// <summary>True if any field is unknown</summary>
    public bool HasUnknownFields => UnknownFieldNames.Count > 0;
}

/// <summary>
/// ML-based unknown data detection service.
/// Uses fuzzy string matching to determine if imported field values
/// exist in the library (DCI, Labos, Familles, Formes, Voies, Spécialités).
/// Threshold >= 80 = known, < 80 = unknown.
/// </summary>
public interface IUnknownDataDetectionService
{
    /// <summary>
    /// Detect unknown fields in a single EditionRow.
    /// Checks: Dci, DciAssociation, Labo, Fam1/2/3, Forme, Voie, Specialite.
    /// </summary>
    Task<DetectionReport> DetectAsync(EditionRow row);

    /// <summary>
    /// Detect unknown fields for a batch of rows (more efficient — loads library once).
    /// Returns one DetectionReport per row, in the same order.
    /// </summary>
    Task<List<DetectionReport>> DetectBatchAsync(IEnumerable<EditionRow> rows);

    /// <summary>
    /// Check if a single value matches any entry in the given library list.
    /// Returns the FieldDetectionResult with score and best match.
    /// </summary>
    FieldDetectionResult CheckValue(string fieldName, string value, IEnumerable<string> libraryValues);
}
```

```csharp
using System.Collections.Generic;
using System.Threading.Tasks;
using AVCNDB.WPF.Models;

namespace AVCNDB.WPF.Contracts.Services;

/// <summary>
/// Source type for an edition file import
/// </summary>
public enum EditionSourceType
{
    ExcelSimple = 0,      // Table Excel simple
    MedicAVICENNA = 1,    // Fichier medic AVICENNA / DBF
    VidalTN = 2,          // VIDAL TN / DBF
    CNAM = 3              // CNAM / XLS
}

/// <summary>
/// Result of an import operation
/// </summary>
public class ImportResult
{
    public bool Success { get; set; }
    public string? ErrorMessage { get; set; }
    public List<EditionRow> Rows { get; set; } = new();
    public int TotalRows => Rows.Count;
    public int UnknownRowCount => Rows.Count(r => r.HasUnknownFields);
}

/// <summary>
/// Service for the Fichier d'édition workflow.
/// Handles Excel import, library validation, row approval/rejection, and export.
/// </summary>
public interface IEditionFileService
{
    /// <summary>
    /// Parse an Excel file and return a list of EditionRows.
    /// Does NOT run detection — call ValidateAgainstLibraryAsync separately.
    /// </summary>
    Task<ImportResult> ImportExcelAsync(string filePath, EditionSourceType sourceType = EditionSourceType.ExcelSimple);

    /// <summary>
    /// Run unknown-data detection on all rows using IUnknownDataDetectionService.
    /// Sets each row's UnknownFields list and ActionFlag (AjouterNew if unknowns found).
    /// Returns the same list with flags populated.
    /// </summary>
    Task<List<EditionRow>> ValidateAgainstLibraryAsync(List<EditionRow> rows);

    /// <summary>
    /// Approve a row: add any unknown field values to the appropriate library tables
    /// (Dci, Labos, Families, Formes, Voies, Specialites) and optionally insert/update
    /// the corresponding Medic record.
    /// </summary>
    Task<bool> ApproveRowAsync(EditionRow row, bool insertMedic = true);

    /// <summary>
    /// Reject a row: mark it as Desaffecte, clear UnknownFields, do not touch library.
    /// </summary>
    Task RejectRowAsync(EditionRow row);

    /// <summary>
    /// Export the current edition file rows to a new Excel file.
    /// </summary>
    Task ExportEditionFileAsync(IEnumerable<EditionRow> rows, string outputPath);

    /// <summary>
    /// Persist a new EditionFileSession record to the database for audit tracking.
    /// </summary>
    Task<EditionFileSession> SaveSessionAsync(
        string sourceFilePath,
        string sourceType,
        int totalRows,
        int unknownRows);
}
```
